# booking/.htaccess (Edwin approved 23 Nov 2026 8:45pm SGT). Only for /booking/.
# The root .htaccess sets X-Frame-Options SAMEORIGIN and CSP frame-ancestors 'self' with "Header always set",
# which blocks the booking popup/iframe on client domains. PHP sends its own per-client CSP and Cache-Control.
<IfModule mod_headers.c>
  Header always unset X-Frame-Options
  Header always unset Content-Security-Policy
</IfModule>
<IfModule mod_expires.c>
  ExpiresActive Off
</IfModule>
# Never serve stray dumps or backups from this folder.
<FilesMatch "\.(sql|gz|bak|log|txt|md|json)$">
  Require all denied
</FilesMatch>
